  '(centaur-tabs-selected :background "#333333" :foreground "#f8f8f2" :weight bold :height 0.9)
  '(centaur-tabs-unselected-modified :background "#181915" :foreground "#fd971f" :height 0.9)
  '(centaur-tabs-selected-modified :background "#333333" :foreground "#fd971f" :weight bold :height 0.9)

  '(header-line :background "#181915" :foreground "#75715e" :box nil :underline nil :overline nil)
  '(tab-line :background "#181915" :foreground "#75715e" :box nil :underline nil :overline nil :height 0.9))

(after! centaur-tabs
  (setq centaur-tabs-style "rounded"          ; GUI: yuvarlak kenar
        centaur-tabs-height 16                ; ince
        centaur-tabs-set-bar 'under            ; aktif: turuncu alt çizgi
        centaur-tabs-set-icons nil
        centaur-tabs-set-close-button nil      ; x yok → sade
        centaur-tabs-set-modified-marker t
        centaur-tabs-modified-marker "·"
        centaur-tabs-cycle-scope 'tabs
        centaur-tabs-label-fixed-length 12
        centaur-tabs-buffer-groups-function #'centaur-tabs-projectile-buffer-groups)
  (unless (display-graphic-p)
    (setq centaur-tabs-style "bar"            ; TTY: XPM radius yok
          centaur-tabs-height 14
          centaur-tabs-modified-marker "*"))
  (centaur-tabs-headline-match)               ; boş alan da koyu
  (map! :n "g t" #'centaur-tabs-forward
        :n "g T" #'centaur-tabs-backward
        ;; Sekmeyi kapat (buffer kill) — vim :bd / :tabclose
        :n "g c" #'kill-current-buffer
        :n "g C" #'centaur-tabs-kill-all-buffers-in-current-group)
  (map! :leader
        :desc "Next tab"     "]" #'centaur-tabs-forward
        :desc "Previous tab" "[" #'centaur-tabs-backward)
  ;; SPC b c = close tab (SPC b d / SPC b k zaten kill-buffer)
  (map! :leader
        (:prefix "b"
         :desc "Close tab (kill buffer)" "c" #'kill-current-buffer)))
#+end_src

* C/C++ Development Setup
#+begin_src emacs-lisp
(use-package! lsp-mode
  :hook ((c-mode c++-mode c-ts-mode c++-ts-mode) . lsp-deferred)
  :custom
  (lsp-clients-clangd-args
   '("--clang-tidy"
     "--completion-style=detailed"
     "--header-insertion=iwyu"
     "--inlay-hints"
     "--function-arg-placeholders=0"
     "--log=error"))
  (lsp-enable-symbol-highlighting t)
  ;; Semantic tokens KAPALI: clangd token'ları treesit level-4'ün üstüne
  ;; lsp-face-semhl-* ile tekrar boyuyordu (operator yeşil, member beyaz vb.) =>
  ;; LSP açılınca renkler kayıyordu.  Tek highlighter treesit kalsın = nvim ile
  ;; birebir, renk sabit.  Diğer LSP özellikleri (hover/def/diag/inlay) etkilenmez.
  (lsp-semantic-tokens-enable nil)
  (lsp-inlay-hint-enable t)
  :config
  (setq lsp-signature-auto-activate t
        lsp-signature-render-documentation t
        lsp-eldoc-render-all t)
  (add-hook 'lsp-mode-hook 'lsp-ui-mode))

(after! lsp-inlay-hints
  (setq lsp-inlay-hints-mode t)
  (custom-set-faces
   '(lsp-inlay-hint-face
     ((t (:foreground "#727072" :background "#2D2A2E" :slant italic :height 0.9))))))

(after! lsp-ui
  (setq lsp-ui-sideline-enable t
        lsp-ui-sideline-show-hover t
        lsp-ui-sideline-show-code-actions t
        lsp-ui-sideline-show-diagnostics t
        lsp-ui-sideline-show-symbol t
        lsp-ui-doc-enable t
        lsp-ui-doc-show-with-cursor t
        lsp-ui-doc-delay 0.15
        lsp-ui-doc-position 'top)
  (custom-set-faces
   '(lsp-ui-doc-background ((t (:background "#2D2A2E"))))
   '(lsp-ui-doc-header ((t (:foreground "#FF6188" :weight bold))))))

(after! treesit
  (setq treesit-font-lock-level 4
        treesit-auto-install-grammar 'always)
  (dolist (dir (list (expand-file-name "tree-sitter" doom-emacs-dir)
                     (expand-file-name ".local/etc/tree-sitter" doom-emacs-dir)
                     (expand-file-name "tree-sitter" user-emacs-directory)))
    (when (file-directory-p dir)
      (add-to-list 'treesit-extra-load-path dir))))

;; C/C++: cc +tree-sitter modülü remap yapar.
;; ESKİ HATA: .cpp c-ts-mode'a (C grammar) bağlıydı => renklendirme bozuk.
(use-package! c-ts-mode
  :defer t
  :config (setq c-ts-mode-indent-offset 2))

(use-package! c++-ts-mode
  :defer t
  :config (setq c-ts-mode-indent-offset 2))

(dolist (pair '(("\\.hpp\\'" . c++-mode)
                ("\\.hh\\'"  . c++-mode)
                ("\\.hxx\\'" . c++-mode)
                ("\\.cc\\'"  . c++-mode)
                ("\\.cxx\\'" . c++-mode)
                ("\\.c\\+\\+\\'" . c++-mode)))
  (add-to-list 'auto-mode-alist pair))

;; clang-format binary yolu (format-on-save artık apheleia / :editor format +onsave)
(use-package! clang-format
  :defer t
  :init
  (setq clang-format-executable
        (or (executable-find "clang-format")
            (executable-find "clang-format-18")
            (expand-file-name "~/.local/bin/clang-format"))))
